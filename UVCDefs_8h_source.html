158 bool hasInterval(uint32_t interval) const {
159 for (int j = 0; j < intervalCount; j++)
160 if (intervals[j] == interval) return true;
161 return false;
162 }
164 uint32_t closestInterval(uint32_t interval) const {
165 if (intervalCount == 0) return 0;
166 uint32_t best = intervals[0];
167 uint32_t bestDiff = 0xFFFFFFFF;
168 for (int j = 0; j < intervalCount; j++) {
169 uint32_t d = intervals[j] > interval ? intervals[j] - interval
170 : interval - intervals[j];
171 if (d < bestDiff) {
172 bestDiff = d;
173 best = intervals[j];
174 }
175 }
176 return best;
177 }
178 uint32_t minInterval() const {
179 uint32_t r = intervals[0];
180 for (int j = 1; j < intervalCount; j++)
181 if (intervals[j] < r) r = intervals[j];
182 return r;
183 }
184 uint32_t maxInterval() const {
185 uint32_t r = intervals[0];
186 for (int j = 1; j < intervalCount; j++)
187 if (intervals[j] > r) r = intervals[j];
188 return r;
189 }

138 }
139 const uint32_t gen = uvc_.streamGeneration();
140 if (gen != generation_) {

146 const uint32_t now = platform::micros();
147 if (sentOnce_ && now - lastFrameUs_ < intervalUs) return false;
148 if (!uvc_.isReady()) return false;

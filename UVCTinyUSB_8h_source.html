45inline bool xfer(uint8_t ep, uint8_t *buf, uint16_t len) {
46 const uint8_t port = rhport();
47 if (!usbd_edpt_claim(port, ep)) return false;
48 if (!callXfer(&usbd_edpt_xfer, port, ep, buf, len)) {

67inline uint16_t drvOpen(uint8_t port, tusb_desc_interface_t const *itf,
68 uint16_t maxLen) {
69 UVCDevice *dev = UVCDevice::instance();
70 if (dev == nullptr) return 0;
71 if (itf->bInterfaceClass != uvc_spec::CC_VIDEO ||
72 itf->bInterfaceSubClass != uvc_spec::SC_VIDEOCONTROL ||
73 itf->bInterfaceNumber != dev->controlInterface())
74 return 0;
75 const uint8_t *p = (const uint8_t *)itf;
76 const uint8_t *end = p + maxLen;
77 uint16_t len = 0;
78 uint16_t mps = 0;
79 bool epOpen = false;
80 while (p + 2 <= end) {
81 const uint8_t bLength = p[0], type = p[1];
82 if (bLength < 2 || p + bLength > end) break;
83 if (len > 0) {
84 if (type == uvc_spec::DESC_IAD) break;
85 if (type == uvc_spec::DESC_INTERFACE && p[2] != dev->controlInterface() &&
86 p[2] != dev->streamingInterface())
87 break;
88 }
89 if (type == uvc_spec::DESC_ENDPOINT && p[2] == dev->endpoint()) {
90 if (!usbd_edpt_open(port, (tusb_desc_endpoint_t const *)p)) {
91 TINYUVC_LOGE("could not open endpoint 0x%02x", p[2]);
92 return 0;
93 }
94 mps = (uint16_t)(le::get16(p + 4) & 0x7FF);
95 epOpen = true;
96 }
97 p += bLength;
98 len = (uint16_t)(len + bLength);
99 }
100 if (!epOpen) {
101 TINYUVC_LOGE("no streaming endpoint found in descriptor");
102 return 0;
103 }
104 rhport() = port;
105 dev->usbOpened(&xfer, mps);
106 return len;
107}

109inline bool drvControl(uint8_t port, uint8_t stage,
110 tusb_control_request_t const *request) {
111 UVCDevice *dev = UVCDevice::instance();
112 if (dev == nullptr) return false;
113 UVCRequest r;
114 r.bmRequestType = *(const uint8_t *)request;
115 r.bRequest = request->bRequest;
116 r.wValue = request->wValue;
117 r.wIndex = request->wIndex;
118 r.wLength = request->wLength;
119
120 if (stage == CONTROL_STAGE_SETUP) {
121 uint8_t *data = nullptr;
122 uint16_t len = 0;
123 switch (dev->handleControlSetup(r, data, len)) {
124 case UVCControlResult::DataIn:
125 case UVCControlResult::DataOut:
126 return tud_control_xfer(port, request, data, len);
127 case UVCControlResult::Status:
128 return tud_control_status(port, request);
129 case UVCControlResult::Handled:
130 return true;
131 case UVCControlResult::Stall:
132 default:
133 return false;
134 }
135 }
136 if (stage == CONTROL_STAGE_DATA && !r.isIn() && r.wLength > 0) {
137 return dev->handleControlData(r, r.wLength);
138 }
139 return true;
140}

121 bool addResolution(UVCFormat format, uint16_t width, uint16_t height,
122 const uint8_t *fpsList, size_t fpsCount,
123 uint32_t maxFrameSize = 0) {
124 if (active_) {
125 TINYUVC_LOGE("addResolution() must be called before begin()");
126 return false;
127 }
128 bool anyFps = false;
129 for (size_t i = 0; i < fpsCount; i++) anyFps |= fpsList[i] != 0;
130 if (width == 0 || height == 0 || !anyFps) {
131 TINYUVC_LOGE("addResolution(): invalid size or no frame rate");
132 return false;
133 }
134 if (format == UVCFormat::YUY2 && (width & 1)) {
135 TINYUVC_LOGE("addResolution(): YUY2 needs an even width");
136 return false;
137 }
138 UVCFormatSpec *f = findFormat(format);
139 if (f == nullptr) {
140 if (formatCount_ >= TINYUVC_MAX_FORMATS) {
141 TINYUVC_LOGE("too many formats (TINYUVC_MAX_FORMATS)");
142 return false;
143 }
144 f = &formats_[formatCount_++];
145 *f = UVCFormatSpec();
146 f->format = format;
147 }
148 if (f->frameCount >= TINYUVC_MAX_FRAMES) {
149 TINYUVC_LOGE("too many resolutions (TINYUVC_MAX_FRAMES)");
150 return false;
151 }
152 UVCFrameSpec &fr = f->frames[f->frameCount];
153 fr = UVCFrameSpec();
154 fr.width = width;
155 fr.height = height;
156 const uint32_t raw = (uint32_t)width * height * 2;
157 fr.maxFrameSize = format == UVCFormat::YUY2 ? raw
158 : maxFrameSize != 0 ? maxFrameSize
159 : raw;
160 for (size_t i = 0; i < fpsCount; i++) {
161 if (fpsList[i] == 0 || fr.intervalCount >= TINYUVC_MAX_INTERVALS) continue;
162 fr.intervals[fr.intervalCount++] = fpsToInterval(fpsList[i]);
163 }
164 f->frameCount++;
165 return true;
166 }

198 bool begin() {
199 if (active_) return true;
200 if (formatCount_ == 0) {
201 TINYUVC_LOGE("begin(): no resolution defined - call addResolution()");
202 return false;
203 }
204 UVCDevice *&inst = instance();
205 if (inst != nullptr && inst != this) {
206 TINYUVC_LOGE("begin(): only one UVCDevice can be active");
207 return false;
208 }
209 maxPacketSize_ = platform::maxPacketSize();
210 payloadSize_ = effectivePayloadSize(payloadSizeCfg_, maxPacketSize_);
211 buffer_ = platform::allocBuffer(payloadSize_);
212 if (buffer_ == nullptr) {
213 TINYUVC_LOGE("begin(): could not allocate %u bytes", (unsigned)payloadSize_);
214 return false;
215 }
216 probe_ = negotiate(UVCProbeCommit());
217 commit_ = probe_;
218 inst = this;
219 active_ = true;
220 if (!platform::begin(*this)) {
221 TINYUVC_LOGE("begin(): USB registration failed");
222 active_ = false;
223 inst = nullptr;
224 platform::freeBuffer(buffer_);
225 buffer_ = nullptr;
226 return false;
227 }
228 TINYUVC_LOGI("UVC device started: %d format(s), payload %u, packet %u",
229 formatCount_, (unsigned)payloadSize_, (unsigned)maxPacketSize_);
230 return true;
231 }

278 bool writeFrame(const uint8_t *data, size_t len, PixelFormat fmt) {
279 if (!isReady()) return false;
280 if (data == nullptr || len == 0) return false;
281 const UVCStreamSettings s = settings_;
282 if (s.format == UVCFormat::MJPEG) {
283 if (fmt != PixelFormat::MJPEG) {
284 TINYUVC_LOGW("writeFrame(): host expects MJPEG, got %s", toString(fmt));
285 dropped_++;
286 return false;
287 }
288 if (len > maxFrameSize_) {
289 TINYUVC_LOGW("writeFrame(): JPEG of %u bytes exceeds max frame size %u",
290 (unsigned)len, (unsigned)maxFrameSize_);
291 dropped_++;
292 return false;
293 }
294 } else {
295 const size_t expected = (size_t)s.width * s.height * bytesPerPixel(fmt);
296 if (fmt == PixelFormat::MJPEG || len != expected) {
297 TINYUVC_LOGW("writeFrame(): host expects %ux%u YUY2: got %u bytes %s",
298 s.width, s.height, (unsigned)len, toString(fmt));
299 dropped_++;
300 return false;
301 }
302 }
303 frameData_ = data;
304 frameLen_ = len;
305 frameFmt_ = fmt;
306 framePts_ = platform::micros();
307 framePending_.store(true);
308 platform::AppLock lock;
309 kick();
310 return true;
311 }

395 UVCControlResult handleControlSetup(const UVCRequest &r, uint8_t *&data,
396 uint16_t &len) {
397 data = ctrlBuf_;
398 len = 0;
399 if (r.type() == UVCRequest::Standard) return handleStandard(r, data, len);
400 if (r.type() != UVCRequest::Class || r.recipient() != UVCRequest::Interface)
401 return stall(uvc_spec::ERR_INVALID_REQUEST);
402
403 const uint8_t itf = (uint8_t)(r.wIndex & 0xFF);
404 const uint8_t entity = (uint8_t)(r.wIndex >> 8);
405 const uint8_t selector = (uint8_t)(r.wValue >> 8);
406
407 if (itf == controlInterface()) {
408 if (entity == 0 && selector == uvc_spec::VC_REQUEST_ERROR_CODE_CONTROL) {
409 if (r.bRequest == uvc_spec::GET_CUR) return reply(&errorCode_, 1, r, data, len);
410 if (r.bRequest == uvc_spec::GET_INFO) {
411 static const uint8_t info = 0x01; // GET supported
412 return reply(&info, 1, r, data, len);
413 }
414 }
415 // the camera and output terminal have no controls
416 return stall(entity == 0 ? uvc_spec::ERR_INVALID_CONTROL
417 : (entity <= uvc_spec::ID_OUTPUT_TERMINAL
418 ? uvc_spec::ERR_INVALID_CONTROL
419 : uvc_spec::ERR_INVALID_UNIT));
420 }
421
422 if (itf == streamingInterface() && entity == 0 &&
423 (selector == uvc_spec::VS_PROBE_CONTROL || selector == uvc_spec::VS_COMMIT_CONTROL)) {
424 const bool probe = selector == uvc_spec::VS_PROBE_CONTROL;
425 switch (r.bRequest) {
426 case uvc_spec::SET_CUR:
427 if (r.wLength == 0) return stall(uvc_spec::ERR_INVALID_REQUEST);
428 len = r.wLength < sizeof(ctrlBuf_) ? r.wLength : sizeof(ctrlBuf_);
429 errorCode_ = uvc_spec::ERR_NONE;
430 return UVCControlResult::DataOut;
431 case uvc_spec::GET_CUR:
432 return replyProbe(probe ? probe_ : commit_, r, data, len);
433 case uvc_spec::GET_DEF:
434 return replyProbe(negotiate(UVCProbeCommit()), r, data, len);
435 case uvc_spec::GET_MIN:
436 return replyProbe(limit(false), r, data, len);
437 case uvc_spec::GET_MAX:
438 return replyProbe(limit(true), r, data, len);
439 case uvc_spec::GET_RES: {
440 UVCProbeCommit res; // all zero: no resolution steps
441 return replyProbe(res, r, data, len);
442 }
443 case uvc_spec::GET_LEN: {
444 uint8_t l[2];
445 le::put16(l, uvc_spec::PROBE_COMMIT_SIZE);
446 return reply(l, 2, r, data, len);
447 }
448 case uvc_spec::GET_INFO: {
449 const uint8_t info = 0x03; // GET and SET supported
450 return reply(&info, 1, r, data, len);
451 }
452 default:
453 return stall(uvc_spec::ERR_INVALID_REQUEST);
454 }
455 }
456 return stall(itf == streamingInterface() ? uvc_spec::ERR_INVALID_CONTROL
457 : uvc_spec::ERR_INVALID_UNIT);
458 }

462 bool handleControlData(const UVCRequest &r, uint16_t len) {
463 if (r.type() != UVCRequest::Class || r.bRequest != uvc_spec::SET_CUR) return true;
464 const uint8_t itf = (uint8_t)(r.wIndex & 0xFF);
465 const uint8_t selector = (uint8_t)(r.wValue >> 8);
466 if (itf != streamingInterface()) return false;
467 UVCProbeCommit in;
468 memcpy(in.raw, ctrlBuf_,
469 len < uvc_spec::PROBE_COMMIT_SIZE ? len : uvc_spec::PROBE_COMMIT_SIZE);
470 if (selector == uvc_spec::VS_PROBE_CONTROL) {
471 probe_ = negotiate(in);
472 TINYUVC_LOGD("probe: format %d frame %d interval %u", probe_.formatIndex(),
473 probe_.frameIndex(), (unsigned)probe_.frameInterval());
474 return true;
475 }
476 if (selector == uvc_spec::VS_COMMIT_CONTROL) {
477 commit_ = negotiate(in);
478 startStream(commit_);
479 return true;
480 }
481 return false;
482 }

581 UVCControlResult handleStandard(const UVCRequest &r, uint8_t *&data,
582 uint16_t &len) {
583 constexpr uint8_t GET_INTERFACE = 0x0A, SET_INTERFACE = 0x0B;
584 if (r.recipient() == UVCRequest::Endpoint) {
585 // CLEAR_FEATURE(ENDPOINT_HALT): how hosts stop a bulk stream
586 constexpr uint8_t CLEAR_FEATURE = 0x01;
587 if (r.bRequest == CLEAR_FEATURE && (r.wIndex & 0xFF) == endpoint_) {
588 TINYUVC_LOGI("host stopped the stream (clear halt)");
589 stopStream();
590 }
591 return UVCControlResult::Handled;
592 }
593 if (r.recipient() == UVCRequest::Interface) {
594 if (r.bRequest == SET_INTERFACE) {
595 if (r.wValue != 0) return stall(uvc_spec::ERR_INVALID_REQUEST);
596 if ((r.wIndex & 0xFF) == streamingInterface() && isStreaming()) {
597 TINYUVC_LOGI("host stopped the stream (set interface)");
598 stopStream();
599 }
600 return UVCControlResult::Status;
601 }
602 if (r.bRequest == GET_INTERFACE) {
603 const uint8_t alt = 0;
604 return reply(&alt, 1, r, data, len);
605 }
606 }
607 return UVCControlResult::Stall;
608 }

611 UVCProbeCommit negotiate(const UVCProbeCommit &in) const {
612 UVCProbeCommit out;
613 uint8_t fi = in.formatIndex();
614 if (fi < 1 || fi > formatCount_) fi = 1;
615 const UVCFormatSpec &fmt = formats_[fi - 1];
616 uint8_t fr = in.frameIndex();
617 if (fr < 1 || fr > fmt.frameCount) fr = 1;
618 const UVCFrameSpec &frame = fmt.frames[fr - 1];
619 uint32_t interval = in.frameInterval() == 0
620 ? frame.intervals[0]
621 : frame.closestInterval(in.frameInterval());
622 out.setHint(in.bmHint());
623 out.setFormatIndex(fi);
624 out.setFrameIndex(fr);
625 out.setFrameInterval(interval);
626 out.setMaxVideoFrameSize(frame.maxFrameSize);
627 out.setMaxPayloadTransferSize(payloadSize_);
628 out.setClockFrequency(UVCDescriptorBuilder::kClockFrequency);
629 out.setFramingInfo(0x03); // FID required, EOF may be present
630 out.setVersions(1, 1, 1);
631 return out;
632 }

635 UVCProbeCommit limit(bool max) const {
636 UVCProbeCommit in;
637 const uint8_t fi = max ? formatCount_ : 1;
638 const UVCFormatSpec &fmt = formats_[fi - 1];
639 const uint8_t fr = max ? fmt.frameCount : 1;
640 in.setFormatIndex(fi);
641 in.setFrameIndex(fr);
642 const UVCFrameSpec &frame = fmt.frames[fr - 1];
643 in.setFrameInterval(max ? frame.maxInterval() : frame.minInterval());

647 void startStream(const UVCProbeCommit &pc) {
648 const UVCFormatSpec &fmt = formats_[pc.formatIndex() - 1];
649 const UVCFrameSpec &fr = fmt.frames[pc.frameIndex() - 1];
650 UVCStreamSettings s;
651 s.format = fmt.format;
652 s.width = fr.width;
653 s.height = fr.height;
654 s.interval = pc.frameInterval();
655 s.formatIndex = pc.formatIndex();
656 s.frameIndex = pc.frameIndex();
657 settings_ = s;
658 maxFrameSize_ = fr.maxFrameSize;
659 generation_.fetch_add(1);
660 streaming_.store(true);
661 TINYUVC_LOGI("host started %s %ux%u @ %u fps", toString(s.format), s.width,
662 s.height, (unsigned)s.fps());
663 if (streamCb_) streamCb_(true, s, streamCbRef_);
664 kick();
665 }

675 void finishFrame(bool complete) {

682
685 bool startTransfer() {
686 const uint32_t gen = generation_.load();
687 if (encoder_.active() && encoderGeneration_ != gen) {
688 // the host restarted or stopped the stream in the middle of a frame
689 encoder_.abort();
690 zlpPending_ = false;
691 finishFrame(false);
692 }
693 if (!streaming_.load()) {
694 zlpPending_ = false;
695 if (framePending_.load() && !encoder_.active()) finishFrame(false);
696 return false;
697 }
698 if (xfer_ == nullptr || buffer_ == nullptr) return false;
699 if (zlpPending_) {
700 zlpPending_ = false;
701 return xfer_(endpoint_, buffer_, 0);
702 }
703 if (!encoder_.active()) {
704 if (!framePending_.load()) return false;
705 fid_ = !fid_;
706 encoder_.begin(frameData_, frameLen_, frameFmt_, fid_, withPts_, framePts_);
707 encoderGeneration_ = gen;
708 }
709 size_t len = encoder_.fill(buffer_, payloadSize_);
710 if (!encoder_.active()) {
711 // last payload: the frame data is no longer needed
712 finishFrame(true);
713 // a transfer that ends on a packet boundary needs a zero length packet,
714 // or the host keeps waiting for the rest of the payload
715 zlpPending_ = len < payloadSize_ && (len % maxPacketSize_) == 0;
716 }
717 if (xfer_(endpoint_, buffer_, (uint16_t)len)) return true;
718 TINYUVC_LOGW("transfer failed");
719 zlpPending_ = false;
720 if (encoder_.active()) {
721 encoder_.abort();
722 finishFrame(false);
723 }
724 return false;
725 }
726
728 void kick() {
729 for (int i = 0; i < 3; i++) {
730 bool expected = false;
731 if (!inFlight_.compare_exchange_strong(expected, true)) return;
732 if (startTransfer()) return;
733 inFlight_.store(false);
734 // something might have arrived while we owned inFlight_
735 if (!(framePending_.load() && streaming_.load())) return;
736 }
737 }

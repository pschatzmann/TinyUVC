25inline bool toPixelFormat(audio_tools::VideoFormat format, PixelFormat &out) {
26 using audio_tools::VideoFormat;
27 switch (format) {
28 case VideoFormat::MJPEG:
29 out = PixelFormat::MJPEG;
30 return true;
31 case VideoFormat::YUV422:
32 out = PixelFormat::YUY2;
33 return true;
34 case VideoFormat::RGB565:
35 out = PixelFormat::RGB565_LE;
36 return true;
37 case VideoFormat::RGB888:
38 out = PixelFormat::RGB888;
39 return true;
40 case VideoFormat::RAW: // 24 bit BGR
41 out = PixelFormat::BGR888;
42 return true;
43 default:
44 return false; // H264/MPEG need a decoder first
45 }
46}

50inline bool addResolution(UVCDevice &uvc, const audio_tools::VideoInfo &info,
51 uint32_t maxJpegSize = 0) {
52 PixelFormat pf;
53 if (!info || !toPixelFormat(info.format, pf)) return false;
54 const uint8_t fps = info.fps <= 0 ? 15 : info.fps >= 255 ? 255 : (uint8_t)(info.fps + 0.5f);
55 const bool jpeg = pf == PixelFormat::MJPEG;
56 return uvc.addResolution(jpeg ? UVCFormat::MJPEG : UVCFormat::YUY2, info.width,
57 info.height, fps, jpeg ? maxJpegSize : 0);
58}

132 size_t write(const uint8_t *data, size_t len) override {
133 UVCFrameBuffer &buf = buffers_[fill_];
134 const bool complete = flushOnWrite_ || (buf.empty() && isCompleteJpeg(data, len));
135 if (overflow_ || buf.size() + len > maxFrameSize_) {
136 overflow_ = true;
137 } else if (!buf.append(data, len)) {
138 TINYUVC_LOGW("UVCVideoOutput: out of memory (frame of %u bytes)",
139 (unsigned)(buf.size() + len));
140 overflow_ = true;
141 }

147 void flush() override {
148 UVCFrameBuffer &buf = buffers_[fill_];
149 if (buf.empty() && !overflow_) return; // e.g. flush() after a complete JPEG
150 hadOutput_ = false;
151 if (buf.empty()) {
152 overflow_ = false;
153 dropped_++;
154 return;
155 }
156 bool ok = !overflow_ && uvc_.isStreaming();
157 if (ok && pacing_ && sent_) {
158 const uint32_t intervalUs = uvc_.settings().interval / 10;
159 while (uvc_.isStreaming() && platform::micros() - lastUs_ < intervalUs)
160 platform::idle();
161 }
162 if (ok && !uvc_.isReady()) {
163 const uint32_t timeout = timeoutMs_ > 0 ? timeoutMs_ : (pacing_ ? 1000 : 0);
164 ok = timeout > 0 && uvc_.waitReady(timeout);
165 }
166 if (ok && uvc_.writeFrame(buf.data(), buf.size(), format_)) {
167 // keep this buffer until it is sent, write the next frame to the other
168 fill_ = 1 - fill_;
169 hadOutput_ = true;
170 frames_++;
171 sent_ = true;
172 lastUs_ = platform::micros();
173 } else {
174 dropped_++;
175 }
176 buffers_[fill_].clear();
177 overflow_ = false;
178 }

244
248 bool copy() {
249 if (!uvc_.isReady()) return false;
250 const uint32_t now = platform::micros();
251 if (sent_ && now - lastUs_ < uvc_.settings().interval / 10) return false;
252 size_t len = 0;
253 // the previous frame stays valid until this call: it was sent completely
254 const uint8_t *data = source_.nextFrame(len);
255 if (data == nullptr || len == 0) return false;
256 if (!uvc_.writeFrame(data, len, format_)) return false;
257 lastUs_ = now;
258 sent_ = true;
259 return true;
260 }

262 protected:

264 audio_tools::VideoFrameSource &source_;
265 PixelFormat format_ = PixelFormat::MJPEG;
266 uint32_t lastUs_ = 0;
267 bool sent_ = false;

220 bool control(const UVCRequest &r, const uint8_t *out, uint16_t outLen,
221 std::vector<uint8_t> *in) {
222 UVCSimBus &b = simBus();
223 b.resetControl();
224 tusb_control_request_t req;
225 req.bmRequestType = r.bmRequestType;
226 req.bRequest = r.bRequest;
227 req.wValue = r.wValue;
228 req.wIndex = r.wIndex;
229 req.wLength = r.isIn() ? r.wLength : outLen;
230 bool ok = driver_->control_xfer_cb(0, CONTROL_STAGE_SETUP, &req);
231 if (!ok) {
232 // the TinyUSB core answers GET/SET_INTERFACE itself if the driver
233 // does not
234 return r.type() == UVCRequest::Standard &&
235 r.recipient() == UVCRequest::Interface &&
236 (r.bRequest == 0x0A || r.bRequest == 0x0B);
237 }
238 if (b.ctrlXfer) {
239 if (r.isIn()) {
240 uint16_t n = b.ctrlLen < r.wLength ? b.ctrlLen : r.wLength;
241 if (in) in->assign(b.ctrlBuf, b.ctrlBuf + n);
242 } else {
243 if (b.ctrlLen < outLen) return false;
244 memcpy(b.ctrlBuf, out, outLen);
245 }
246 if (req.wLength > 0 &&
247 !driver_->control_xfer_cb(0, CONTROL_STAGE_DATA, &req))
248 return false;
249 }
250 driver_->control_xfer_cb(0, CONTROL_STAGE_ACK, &req);
251 return true;
252 }

255 bool getProbe(uint8_t request, bool commit, UVCProbeCommit &pc) {
256 std::vector<uint8_t> data;
257 UVCRequest r;
258 r.bmRequestType = 0xA1; // in, class, interface
259 r.bRequest = request;
260 r.wValue = (uint16_t)((commit ? uvc_spec::VS_COMMIT_CONTROL : uvc_spec::VS_PROBE_CONTROL) << 8);
261 r.wIndex = dev_->streamingInterface();
262 r.wLength = uvc_spec::PROBE_COMMIT_SIZE;
263 if (!control(r, nullptr, 0, &data)) return false;
264 if (data.size() != uvc_spec::PROBE_COMMIT_SIZE) return fail("short probe reply");
265 memcpy(pc.raw, data.data(), data.size());
266 return true;
267 }

345 bool buildConfiguration(UVCDevice &dev) {
346 // what a board with USB serial + camera reports
347 static const uint8_t cdc[] = {
348 8, 0x0B, 0, 2, 0x02, 0x02, 0x00, 0, // IAD
349 9, 0x04, 0, 0, 1, 0x02, 0x02, 0x00, 0, // CDC control interface
350 5, 0x24, 0x00, 0x20, 0x01, // header
351 5, 0x24, 0x01, 0x00, 1, // call management
352 4, 0x24, 0x02, 0x02, // ACM
353 5, 0x24, 0x06, 0, 1, // union
354 7, 0x05, 0x81, 0x03, 8, 0, 16, // notification endpoint
355 9, 0x04, 1, 0, 2, 0x0A, 0x00, 0x00, 0, // CDC data interface
356 7, 0x05, 0x02, 0x02, 64, 0, 0, // bulk out
357 7, 0x05, 0x82, 0x02, 64, 0, 0, // bulk in
358 };
359 const uint8_t firstItf = withCdc_ ? 2 : 0;
360 const uint8_t ep = withCdc_ ? 0x83 : 0x81;
361 mps_ = platform::maxPacketSize();
362 uint16_t uvcLen = dev.descriptorLength();
363 config_.assign(9, 0);
364 if (withCdc_) config_.insert(config_.end(), cdc, cdc + sizeof(cdc));
365 size_t off = config_.size();
366 config_.resize(off + uvcLen);
367 uint16_t len = dev.buildDescriptor(config_.data() + off, uvcLen, firstItf,
368 ep, mps_, 4);
369 if (len != uvcLen) return fail("descriptor length mismatch");
370 config_[0] = 9;
371 config_[1] = 0x02;
372 le::put16(&config_[2], (uint16_t)config_.size());
373 config_[4] = (uint8_t)(firstItf + 2); // bNumInterfaces
374 config_[5] = 1; // bConfigurationValue
375 config_[6] = 0;
376 config_[7] = 0x80;
377 config_[8] = 250;
378 return true;
379 }

382 bool parseConfiguration() {
383 formats_.clear();
384 const uint8_t *d = config_.data();
385 const size_t n = config_.size();
386 if (le::get16(d + 2) != n) return fail("config wTotalLength wrong");
387 // check that the descriptors exactly tile the configuration
388 size_t pos = 0;
389 int interfaces = 0;
390 while (pos < n) {
391 if (d[pos] < 2 || pos + d[pos] > n) return fail("bad bLength in configuration");
392 if (d[pos + 1] == uvc_spec::DESC_INTERFACE && d[pos + 3] == 0) interfaces++;
393 pos += d[pos];
394 }
395 if (interfaces != d[4]) return fail("bNumInterfaces wrong");
396
397 // find our IAD
398 pos = 9;
399 size_t iad = 0;
400 while (pos < n) {
401 if (d[pos + 1] == uvc_spec::DESC_IAD && d[pos + 4] == uvc_spec::CC_VIDEO) {
402 iad = pos;
403 break;
404 }
405 pos += d[pos];
406 }
407 if (!iad) return fail("no video interface association");
408 if (d[iad + 5] != uvc_spec::SC_VIDEO_INTERFACE_COLLECTION || d[iad + 3] != 2)
409 return fail("bad video IAD");
410 vcItf_ = d[iad + 2];
411 pos = iad + d[iad];
412 // VideoControl interface
413 if (d[pos + 1] != uvc_spec::DESC_INTERFACE || d[pos + 2] != vcItf_ ||
414 d[pos + 5] != uvc_spec::CC_VIDEO || d[pos + 6] != uvc_spec::SC_VIDEOCONTROL)
415 return fail("IAD not followed by VideoControl interface");
416 pos += d[pos];
417 if (d[pos + 1] != uvc_spec::CS_INTERFACE || d[pos + 2] != uvc_spec::VC_HEADER)
418 return fail("no VC header");
419 if (le::get16(d + pos + 3) != uvc_spec::BCD_UVC) return fail("bcdUVC is not 1.10");
420 const uint16_t vcTotal = le::get16(d + pos + 5);
421 const uint8_t vsItf = d[pos + 12];
422 if (d[pos + 11] != 1 || vsItf != vcItf_ + 1) return fail("VC header baInterfaceNr wrong");
423 size_t vcEnd = pos;
424 while (vcEnd < n && d[vcEnd + 1] == uvc_spec::CS_INTERFACE) vcEnd += d[vcEnd];
425 if (vcEnd - pos != vcTotal) return fail("VC header wTotalLength wrong");
426 pos = vcEnd;
427 // VideoStreaming interface
428 if (d[pos + 1] != uvc_spec::DESC_INTERFACE || d[pos + 2] != vsItf ||
429 d[pos + 6] != uvc_spec::SC_VIDEOSTREAMING || d[pos + 4] != 1)
430 return fail("no VideoStreaming interface with one endpoint");
431 pos += d[pos];
432 if (d[pos + 1] != uvc_spec::CS_INTERFACE || d[pos + 2] != uvc_spec::VS_INPUT_HEADER)
433 return fail("no VS input header");
434 const uint8_t numFormats = d[pos + 3];
435 const uint16_t vsTotal = le::get16(d + pos + 4);
436 ep_ = d[pos + 6];
437 if (d[pos + 8] != uvc_spec::ID_OUTPUT_TERMINAL) return fail("bTerminalLink wrong");
438 if (d[pos] != 13 + numFormats * d[pos + 12]) return fail("VS header bLength wrong");
439 size_t vsStart = pos, vsEnd = pos;
440 while (vsEnd < n && d[vsEnd + 1] == uvc_spec::CS_INTERFACE) vsEnd += d[vsEnd];
441 if (vsEnd - vsStart != vsTotal) return fail("VS header wTotalLength wrong");
442 // formats and frames
443 pos += d[pos];
444 HostFormat *cur = nullptr;
445 int framesLeft = 0;
446 while (pos < vsEnd) {
447 const uint8_t sub = d[pos + 2];
448 if (sub == uvc_spec::VS_FORMAT_MJPEG || sub == uvc_spec::VS_FORMAT_UNCOMPRESSED) {
449 if (framesLeft != 0) return fail("bNumFrameDescriptors wrong");
450 HostFormat f;
451 f.index = d[pos + 3];
452 f.format = sub == uvc_spec::VS_FORMAT_MJPEG ? UVCFormat::MJPEG : UVCFormat::YUY2;
453 if (f.index != formats_.size() + 1) return fail("bFormatIndex not sequential");
454 if (sub == uvc_spec::VS_FORMAT_UNCOMPRESSED &&
455 (d[pos] != 27 || memcmp(d + pos + 5, "YUY2", 4) != 0 || d[pos + 21] != 16))
456 return fail("bad YUY2 format descriptor");
457 if (sub == uvc_spec::VS_FORMAT_MJPEG && d[pos] != 11) return fail("bad MJPEG format descriptor");
458 framesLeft = d[pos + 4];
459 formats_.push_back(f);
460 cur = &formats_.back();
461 } else if (sub == uvc_spec::VS_FRAME_MJPEG || sub == uvc_spec::VS_FRAME_UNCOMPRESSED) {
462 if (cur == nullptr || framesLeft <= 0) return fail("unexpected frame descriptor");
463 if ((sub == uvc_spec::VS_FRAME_MJPEG) != (cur->format == UVCFormat::MJPEG))
464 return fail("frame descriptor type does not match format");
465 HostFrame fr;
466 fr.index = d[pos + 3];
467 fr.width = le::get16(d + pos + 5);
468 fr.height = le::get16(d + pos + 7);
469 fr.maxFrameSize = le::get32(d + pos + 17);
470 fr.defaultInterval = le::get32(d + pos + 21);
471 const uint8_t nInt = d[pos + 25];
472 if (nInt == 0 || d[pos] != 26 + 4 * nInt) return fail("bad frame descriptor length");
473 for (int i = 0; i < nInt; i++) fr.intervals.push_back(le::get32(d + pos + 26 + 4 * i));
474 bool hasDefault = false;
475 for (auto iv : fr.intervals) hasDefault |= iv == fr.defaultInterval;
476 if (!hasDefault) return fail("default interval not in the list");
477 if (fr.index != cur->frames.size() + 1) return fail("bFrameIndex not sequential");
478 if (cur->format == UVCFormat::YUY2 &&
479 fr.maxFrameSize != (uint32_t)fr.width * fr.height * 2)
480 return fail("YUY2 dwMaxVideoFrameBufferSize wrong");
481 cur->frames.push_back(fr);
482 framesLeft--;
483 } else if (sub != uvc_spec::VS_COLORFORMAT) {
484 return fail("unknown VS descriptor");
485 }
486 pos += d[pos];
487 }
488 if (framesLeft != 0) return fail("bNumFrameDescriptors wrong");
489 if (formats_.size() != numFormats) return fail("bNumFormats wrong");
490 // bulk endpoint
491 if (pos >= n || d[pos + 1] != uvc_spec::DESC_ENDPOINT || d[pos + 2] != ep_ ||
492 (d[pos + 3] & 3) != 2)
493 return fail("no bulk endpoint after the VS descriptors");
494 if (le::get16(d + pos + 4) != mps_) return fail("wMaxPacketSize wrong");
495 return true;
496 }

501 const uint8_t *d = config_.data();
502 const size_t n = config_.size();
503 size_t pos = 9;

541 void urbComplete(const uint8_t *mem, size_t len, bool shortUrb) {
542 if (len == 0 && headerSize_ == 0) return; // ZLP between payloads
543 payloadSize_ += (uint32_t)len;
544 if (headerSize_ == 0 && !skipPayload_) {
545 int ret = decodeStart(mem, len);
546 if (ret < 0) {
547 skipPayload_ = true;
548 } else {
549 headerSize_ = (uint8_t)ret;
550 memcpy(header_, mem, ret);
551 mem += ret;
552 len -= ret;
553 }
554 }
555 if (!skipPayload_ && frameActive_) decodeData(mem, len);
556 if (shortUrb || payloadSize_ >= maxPayload_) {
557 if (!skipPayload_ && headerSize_ > 0) decodeEnd(header_);
558 resetPayload();
559 }
560 }
561
563 int decodeStart(const uint8_t *data, size_t len) {
564 if (len < 2 || data[0] < 2 || data[0] > len) {
565 droppedPayloads_++;
566 return -1;
567 }
568 payloads_++;
569 const uint8_t flags = data[1];
570 const int fid = flags & uvc_spec::HDR_FID;
571 if (flags & uvc_spec::HDR_ERR) frameError_ = true;
572 // FID toggled while a frame was active: the previous one is complete
573 if (frameActive_ && fid != lastFid_ && !frame_.empty()) completeFrame();
574 if (!frameActive_) {
575 if (fid == lastFid_) {
576 droppedPayloads_++; // out of sync, wait for the next frame
577 return -1;
578 }
579 frameActive_ = true;
580 frameError_ = (flags & uvc_spec::HDR_ERR) != 0;
581 frame_.clear();
582 framePts_ = (flags & uvc_spec::HDR_PTS) && data[0] >= 6;
583 framePtsValue_ = framePts_ ? le::get32(data + 2) : 0;
584 }
585 lastFid_ = fid;
586 return data[0];
587 }
588
590 void decodeData(const uint8_t *mem, size_t len) {
591 if (len == 0) return;
592 if (frame_.size() + len > maxVideoFrameSize_) {
593 frameError_ = true; // overflow: data is dropped
594 len = maxVideoFrameSize_ - frame_.size();
595 }
596 frame_.insert(frame_.end(), mem, mem + len);
597 }
598
600 void decodeEnd(const uint8_t *header) {
601 if ((header[1] & uvc_spec::HDR_EOF) && frameActive_ && !frame_.empty()) completeFrame();
602 }

604 void completeFrame() {
605 // uvc_video_validate_buffer(): uncompressed frames must be complete
606 if (settings_.format == UVCFormat::YUY2 && frame_.size() != maxVideoFrameSize_)
607 frameError_ = true;
608 Frame f;
609 f.data = frame_.data();
610 f.size = frame_.size();
611 f.settings = settings_;
612 f.error = frameError_;
613 f.hasPts = framePts_;
614 f.pts = framePtsValue_;
615 f.index = frames_;
616 frames_++;
617 if (frameError_) errorFrames_++;
618 if (onFrame_) onFrame_(f);
619 frameActive_ = false;
620 frameError_ = false;
621 frame_.clear();
622 }

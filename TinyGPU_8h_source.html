200 bool writeData(tinygpu::ISurface<RGB_T> &surface, size_t x,
201 size_t y) override {
202 if (!uvc_.isStreaming() || x >= width() || y >= height()) return false;
203 if (!work_.reserve(rawBytes())) return false;
204 const size_t bpp = UVCTinyGPUPixel<RGB_T>::bytes;
205 const size_t w = std::min(surface.width(), width() - x);
206 const size_t h = std::min(surface.height(), height() - y);
207 const uint8_t *src = surface.data();
208 for (size_t row = 0; row < h; row++) {
209 memcpy(work_.data() + ((y + row) * width() + x) * bpp,
210 src + row * surface.width() * bpp, w * bpp);
211 }
212 dirty_ = true;

285 static size_t jpegSinkWrite(void *arg, size_t index, const void *data,
286 size_t len) {
287 JpegSink *sink = (JpegSink *)arg;
288 if (data == nullptr) return 0; // end of the image
289 if (index + len > sink->capacity) {
290 sink->overflow = true;
291 return 0; // stops the encoder
292 }
293 memcpy(sink->data + index, data, len);

342 UVCFrameBuffer *nextBuffer(size_t bytes) {
343 UVCFrameBuffer &buf = buffers_[next_];
344 if (buf.capacity() < bytes) {

356 bool sendJpeg(const uint8_t *pixels) {
357 const uint16_t w = (uint16_t)width(), h = (uint16_t)height();
358#if TINYUVC_JPEG_ESP32
359 if (useEsp32Jpeg()) {
360 const size_t maxJpeg = selected().maxFrameSize;
361 UVCFrameBuffer *buf = nextBuffer(maxJpeg);
362 if (buf == nullptr) return false;
363 JpegSink sink{buf->data(), maxJpeg, 0, false};
364 const bool ok = fmt2jpg_cb((uint8_t *)pixels, (size_t)w * h * 2, w, h,
365 PIXFORMAT_RGB565, quality_, jpegSinkWrite,
366 &sink);
367 if (!ok || sink.overflow) {
368 TINYUVC_LOGW("JPEG encoding failed%s",
369 sink.overflow ? ": increase maxJpegSize" : "");
370 dropped_++;
371 return false;
372 }
373 lastJpegSize_ = sink.len;
374 return send(*buf, sink.len, PixelFormat::MJPEG);
375 }
376#endif
377#if TINYUVC_JPEG_TINYJPEG
378 JEPixelFormat format = JE_FMT_RGB888;
379 if (UVCTinyGPUPixel<RGB_T>::bytes == 2) {
380 // TinyGPU's RGB565 is byte-swapped, TinyJPEG reads native order
381 const size_t bytes = (size_t)w * h * 2;
382 if (pixels != work_.data()) {
383 if (!work_.reserve(bytes)) {
384 dropped_++;
385 return false;
386 }
387 }
388 uint8_t *dst = work_.data();
389 for (size_t i = 0; i < bytes; i += 2) {
390 const uint8_t hi = pixels[i];
391 dst[i] = pixels[i + 1];
392 dst[i + 1] = hi;
393 }
394 pixels = dst;
395 format = JE_FMT_RGB565;
396 }
397 const size_t maxJpeg = selected().maxFrameSize;
398 UVCFrameBuffer *buf = nextBuffer(maxJpeg);
399 if (buf == nullptr) return false;
400 if (encoder_ == nullptr) {
401 // the encoder works on its workspace all the time: internal RAM is
402 // much faster than PSRAM (where new would put it on an ESP32)
403 void *mem = UVCMemory::alloc(sizeof(tinyjpeg::TinyJPEGEncoder),
404 UVCMemoryType::RAMOnly);
405 if (mem == nullptr) {
406 dropped_++;
407 return false;
408 }
409 encoder_ = new (mem) tinyjpeg::TinyJPEGEncoder();
410 }
411 encoder_->setQuality(quality_);
412 size_t len = 0;
413 const JERESULT r =
414 encoder_->encodeJpg(pixels, w, h, format, buf->data(), maxJpeg, len);
415 if (r != JER_OK) {
416 TINYUVC_LOGW("JPEG encoding failed (%d): increase maxJpegSize?", (int)r);
417 dropped_++;
418 return false;
419 }
420 lastJpegSize_ = len;
421 return send(*buf, len, PixelFormat::MJPEG);
422#else
423 (void)pixels;
424 (void)w;
425 (void)h;
426 dropped_++;
427 return false;
428#endif
429 }

431 bool send(UVCFrameBuffer &buf, size_t len, PixelFormat format) {
432 if (pacing_ && sent_) {
433 const uint32_t intervalUs = uvc_.settings().interval / 10;
434 while (uvc_.isStreaming() && platform::micros() - lastUs_ < intervalUs)
435 platform::idle();
436 }
437 if (!uvc_.isReady() && !uvc_.waitReady(timeoutMs_)) {
438 dropped_++;
439 return false;
440 }
441 if (!uvc_.writeFrame(buf.data(), len, format)) {
442 dropped_++;
443 return false;
444 }
445 next_ = 1 - next_; // keep this buffer until it is sent
446 lastUs_ = platform::micros();
447 sent_ = true;
448 frames_++;
449 return true;
450 }

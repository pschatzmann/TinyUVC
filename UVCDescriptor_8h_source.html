195 static uint16_t formatBlockLength(const UVCFormatSpec &fmt) {
196 uint16_t len = fmt.format == UVCFormat::MJPEG ? kMjpegFormatLen : kYuy2FormatLen;
197 for (int i = 0; i < fmt.frameCount; i++) len += frameLength(fmt.frames[i]);
198 return (uint16_t)(len + kColorMatchingLen);

201 static void formatBlock(Writer &w, const UVCFormatSpec &fmt, uint8_t index) {
202 const bool mjpeg = fmt.format == UVCFormat::MJPEG;
203 if (mjpeg) {
204 w.u8(kMjpegFormatLen);
205 w.u8(uvc_spec::CS_INTERFACE);
206 w.u8(uvc_spec::VS_FORMAT_MJPEG);
207 w.u8(index);
208 w.u8(fmt.frameCount);
209 w.u8(0); // bmFlags: no fixed size samples
210 w.u8(1); // bDefaultFrameIndex
211 w.u8(0); // bAspectRatioX
212 w.u8(0); // bAspectRatioY
213 w.u8(0); // bmInterlaceFlags
214 w.u8(0); // bCopyProtect
215 } else {
216 static const uint8_t kGuidYUY2[16] = {'Y', 'U', 'Y', '2', 0x00, 0x00,
217 0x10, 0x00, 0x80, 0x00, 0x00, 0xAA,
218 0x00, 0x38, 0x9B, 0x71};
219 w.u8(kYuy2FormatLen);
220 w.u8(uvc_spec::CS_INTERFACE);
221 w.u8(uvc_spec::VS_FORMAT_UNCOMPRESSED);
222 w.u8(index);
223 w.u8(fmt.frameCount);
224 for (uint8_t b : kGuidYUY2) w.u8(b);
225 w.u8(16); // bBitsPerPixel
226 w.u8(1); // bDefaultFrameIndex
227 w.u8(0); // bAspectRatioX
228 w.u8(0); // bAspectRatioY
229 w.u8(0); // bmInterlaceFlags
230 w.u8(0); // bCopyProtect
231 }
232
233 for (int i = 0; i < fmt.frameCount; i++) {
234 const UVCFrameSpec &fr = fmt.frames[i];
235 const uint32_t maxFps = intervalToFps(fr.minInterval());
236 const uint32_t minFps = intervalToFps(fr.maxInterval());
237 w.u8((uint8_t)frameLength(fr));
238 w.u8(uvc_spec::CS_INTERFACE);
239 w.u8(mjpeg ? uvc_spec::VS_FRAME_MJPEG : uvc_spec::VS_FRAME_UNCOMPRESSED);
240 w.u8((uint8_t)(i + 1)); // bFrameIndex
241 w.u8(0); // bmCapabilities
242 w.u16(fr.width);
243 w.u16(fr.height);
244 w.u32(fr.maxFrameSize * 8 * (minFps ? minFps : 1)); // dwMinBitRate
245 w.u32(fr.maxFrameSize * 8 * (maxFps ? maxFps : 1)); // dwMaxBitRate
246 w.u32(fr.maxFrameSize); // dwMaxVideoFrameBufferSize
247 w.u32(fr.intervals[0]); // dwDefaultFrameInterval
248 w.u8(fr.intervalCount); // bFrameIntervalType: discrete
249 for (int j = 0; j < fr.intervalCount; j++) w.u32(fr.intervals[j]);
250 }
251
252 // Color matching: BT.709 primaries / transfer, SMPTE 170M (BT.601) matrix
253 w.u8(kColorMatchingLen);
254 w.u8(uvc_spec::CS_INTERFACE);
255 w.u8(uvc_spec::VS_COLORFORMAT);
256 w.u8(1);
257 w.u8(1);
258 w.u8(4);
259 }

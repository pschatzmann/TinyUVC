106 bool reserve(size_t size) {
107 if (size <= capacity_) return true;
108 uint8_t *p = UVCMemory::alloc(size, type_);
109 if (p == nullptr) return false;
110 if (size_ > 0) memcpy(p, data_, size_);
111 UVCMemory::free(data_);

118 bool append(const uint8_t *data, size_t len) {
119 const size_t needed = size_ + len;
120 if (needed > capacity_) {
121 // grow by 50% to limit the number of copies; fall back to the
122 // exact size if that is too much
123 size_t cap = capacity_ + capacity_ / 2;
124 if (cap < needed) cap = needed;
125 if (cap < 4096) cap = 4096;
126 if (!reserve(cap) && !reserve(needed)) return false;
127 }
128 memcpy(data_ + size_, data, len);
129 size_ = needed;
130 return true;
131 }
